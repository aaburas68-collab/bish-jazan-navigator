',iconSize:[37,37],iconAnchor:[18,18]}); state.carMarker=L.marker(ll,{icon,zIndexOffset:1000}).addTo(map); }else state.carMarker.setLatLng(ll); } function animate(t){ if(state.tripState!=="running")return; if(state.lastFrameTime!==null)state.tripElapsedMs+=t-state.lastFrameTime; state.lastFrameTime=t; const duration=Math.max(20000,Math.min(60000,state.routeDuration*1000)); const p=state.tripElapsedMs/duration;updateProgress(p); if(p>=1){ state.tripState="arrived";stopAnimation(); $("panelTitle").textContent="انتهت المحاكاة"; $("panelSubtitle").textContent="هذا ليس إثبات وصول GPS"; $("pauseTripBtn").disabled=true;$("startTripBtn").style.display="block"; $("startTripBtn").textContent="إعادة المحاكاة";toast("انتهت محاكاة حركة السيارة");return; } state.animationId=requestAnimationFrame(animate); } function startTrip(){ if(!state.routeLine||!state.routeCoords.length){toast("احسب المسار أولًا");return;} state.tripState="running";state.tripElapsedMs=0;state.lastFrameTime=null; $("tripActions").classList.add("show");$("progressWrap").style.display="block"; $("pauseTripBtn").textContent="⏸ إيقاف مؤقت";$("pauseTripBtn").disabled=false; $("startTripBtn").style.display="none";$("panelTitle").textContent="محاكاة الرحلة"; $("panelSubtitle").textContent="الألوان تعتمد على GPS الفعلي، لا على المحاكاة"; updateProgress(0);state.animationId=requestAnimationFrame(animate); }
